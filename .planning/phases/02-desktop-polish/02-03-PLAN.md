**1. Create `src/components/ui/SegmentedControl.tsx`:**

Build a reusable segmented control with Framer Motion `layoutId` sliding pill indicator. This component replaces the plain button group for employment type selection.

```tsx
import { motion } from "framer-motion";

interface SegmentedOption<T extends string> {
  value: T;
  label: string;
}

interface SegmentedControlProps<T extends string> {
  options: SegmentedOption<T>[];
  value: T;
  onChange: (value: T) => void;
  name: string; // unique name for layoutId namespacing — CRITICAL to avoid collision with activeTab in App.tsx
  className?: string;
}

export function SegmentedControl<T extends string>({
  options,
  value,
  onChange,
  name,
  className,
}: SegmentedControlProps<T>) {
  return (
    <div
      className={`flex gap-0.5 p-0.5 bg-muted rounded-lg ${className ?? ""}`}
      role="group"
    >
      {options.map((option) => {
        const isSelected = option.value === value;
        return (
          <button
            key={option.value}
            type="button"
            role="radio"
            aria-checked={isSelected}
            onClick={() => onChange(option.value)}
            className={`
              relative flex-1 px-3 py-1.5 text-sm font-medium rounded-md
              transition-colors duration-100
              focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring
              ${isSelected ? "text-foreground" : "text-muted-foreground hover:text-foreground/80"}
            `}
          >
            {isSelected && (
              <motion.div
                layoutId={`segmented-pill-${name}`}
                className="absolute inset-0 bg-card shadow-sm rounded-[calc(0.5rem-2px)]"
                transition={{
                  type: "spring",
                  stiffness: 500,
                  damping: 35,
                  mass: 0.5,
                }}
              />
            )}
            <span className="relative z-10">{option.label}</span>
          </button>
        );
      })}
    </div>
  );
}
```

Key points:

- `layoutId={`segmented-pill-${name}`}` namespaces the pill to avoid collision with `layoutId="activeTab"` in App.tsx
- `flex-1` makes all options equal width (handles "Contractor", "Retainer", "Employee" labels)
- Spring transition `stiffness: 500, damping: 35` is fast and responsive
- Accessible: uses `role="group"` and `role="radio"` with `aria-checked`

**4. Update AnimatedNumber.tsx — add scale pulse on significant value changes:**

Add a scale pulse animation that fires when the value changes significantly. Use the `key` prop approach for simplicity:

Wrap the existing AnimatedNumber render in a motion.span that pulses. Add this around the return in the AnimatedNumber component:

```tsx
import { useSpring, MotionValue, motion } from "framer-motion";

export function AnimatedNumber({
  value,
  formatter,
  className,
}: AnimatedNumberProps) {
  const spring = useSpring(value, {
    stiffness: 300,
    damping: 30,
    mass: 0.8,
  });

  useEffect(() => {
    spring.set(value);
  }, [value, spring]);

  // Quantize key: only pulse on changes of ~$500+ (prevents excessive animation during typing)
  const pulseKey = Math.round(value / 500);

  return (
    <motion.span
      key={pulseKey}
      className={className}
      initial={{ scale: 1 }}
      animate={{ scale: [1, 1.03, 1] }}
      transition={{ duration: 0.3, ease: "easeInOut" }}
    >
      <AnimatedValue motionValue={spring} formatter={formatter} />
    </motion.span>
  );
}
```

The pulse is subtle (3% scale, not 5%) and only triggers on ~$500 value changes due to `Math.round(value / 500)` quantization. During fast typing, the spring animation handles the visual feedback; the pulse only fires on significant jumps.
